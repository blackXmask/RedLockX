## 🧩 LangGraph Code (Production Ready)

```python
from langgraph.graph import StateGraph, START, END
import requests

class GuardrailState(dict):
    pass


# 🔹 Hybrid Node
def hybrid_layer(state):
    prompt = state["prompt"]

    res = requests.post(
        "https://your-hybrid-space.hf.space/run/predict",
        json={"data": [prompt]}
    ).json()

    state["hybrid"] = res
    return state


# 🔹 ML Node
def ml_layer(state):
    prompt = state["prompt"]

    res = requests.post(
        "https://your-ml-space.hf.space/run/predict",
        json={"data": [prompt]}
    ).json()

    state["ml"] = res
    return state


# 🔹 Decision Node
def decision_layer(state):
    hybrid = state["hybrid"]
    ml = state["ml"]

    hybrid_prob = hybrid["data"][0]["malicious_probability"]
    ml_status = ml["data"][0]["status"]

    is_attack = hybrid_prob > 0.9 or ml_status == "DANGEROUS"

    state["final"] = {
        "verdict": "BLOCK" if is_attack else "ALLOW",
        "risk_score": hybrid_prob * 100,
        "attack_type": ml["data"][0]["attack_type"]["label"],
        "is_safe": not is_attack
    }

    return state


# 🔹 Build Graph
graph = StateGraph(GuardrailState)

graph.add_node("hybrid_layer", hybrid_layer)
graph.add_node("ml_layer", ml_layer)
graph.add_node("decision_layer", decision_layer)

graph.add_edge(START, "hybrid_layer")
graph.add_edge(START, "ml_layer")

graph.add_edge("hybrid_layer", "decision_layer")
graph.add_edge("ml_layer", "decision_layer")

graph.add_edge("decision_layer", END)

app = graph.compile()
```
